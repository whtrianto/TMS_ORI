SELECT TOP 500
    INV_ID AS ID,
    LTRIM(RTRIM(INV_TOOL_ID)) AS TOOL_ID,
    ISNULL(ML_TOOL_DRAW_NO, '') AS TOOL_DRAWING_NO,
    ISNULL(MLR_REV, 0) AS REVISION,
    ISNULL(TC_NAME, '') AS TOOL_NAME,
    CASE INV_STATUS
        WHEN 1 THEN 'New'
        WHEN 2 THEN 'Allocated'
        WHEN 3 THEN 'Available'
        WHEN 4 THEN 'InUsed'
        WHEN 5 THEN 'Onhold'
        WHEN 6 THEN 'Scrapped'
        WHEN 7 THEN 'Repairing'
        WHEN 8 THEN 'Modifying'
        WHEN 9 THEN 'DesignChange'
        ELSE 'Unknown'
    END AS TOOL_STATUS,
    ISNULL(INV_NOTES, '') AS REMARKS
FROM VW_TOOL_INVENTORY
WHERE INV_TOOL_ID IS NOT NULL 
    AND LTRIM(RTRIM(INV_TOOL_ID)) <> ''
    AND INV_STATUS <> 6
ORDER BY INV_TOOL_ID ASC
